find_package(OpenCL REQUIRED)
find_package(Python3 REQUIRED)

set(TARGET_NAME ggml-ocl)

ggml_add_backend_library(${TARGET_NAME}
                         ggml-ocl.cpp
                         ggml-ocl-caps.cpp
                         ggml-ocl-mem.cpp
                         ggml-ocl-kernels.cpp
                         ggml-ocl-exec.cpp
                         ggml-ocl-ops.cpp
                         ops/add.cpp
                         ops/set_rows.cpp
                         ops/get_rows.cpp
                         ops/mul_mat.cpp
                         ops/softmax.cpp
                         ops/rmsnorm.cpp
                         ops/glu.cpp
                         ../../include/ggml-ocl.h)
target_link_libraries(${TARGET_NAME} PRIVATE ${OpenCL_LIBRARIES})
target_include_directories(${TARGET_NAME} PRIVATE ${OpenCL_INCLUDE_DIRS})
target_include_directories(${TARGET_NAME} PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})

option(GGML_OCL_EMBED_KERNELS "Embed OpenCL kernels into the executable" ON)
option(GGML_OCL_PROFILING     "OpenCL profiling (CPU overhead)" OFF)

add_compile_definitions(GGML_OCL_TARGET_VERSION=${GGML_OCL_TARGET_VERSION})

if (GGML_OCL_EMBED_KERNELS)
    add_compile_definitions(GGML_OCL_EMBED_KERNELS)

    set(EMBED_KERNEL_SCRIPT "${CMAKE_CURRENT_SOURCE_DIR}/kernels/embed_kernel.py")
    file(MAKE_DIRECTORY     "${CMAKE_CURRENT_BINARY_DIR}/autogenerated")

    target_include_directories(${TARGET_NAME} PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/autogenerated")
endif ()

if (GGML_OCL_PROFILING)
    add_compile_definitions(GGML_OCL_PROFILING)
endif ()

function(ggml_ocl_add_kernel KNAME)
    set(KERN_HDR ${CMAKE_CURRENT_BINARY_DIR}/autogenerated/${KNAME}.cl.h)
    set(KERN_SRC ${CMAKE_CURRENT_SOURCE_DIR}/kernels/${KNAME}.cl)

    if (GGML_OCL_EMBED_KERNELS)
        message(STATUS "ggml-ocl: embedding kernel ${KNAME}")

        # embed_kernel.py 只写文件不创建目录, 需为每个输出头建目录
        get_filename_component(KERN_HDR_DIR ${KERN_HDR} DIRECTORY)
        file(MAKE_DIRECTORY ${KERN_HDR_DIR})

        # Python must be accessible from command line
        add_custom_command(
            OUTPUT ${KERN_HDR}
            COMMAND ${Python3_EXECUTABLE} ${EMBED_KERNEL_SCRIPT} ${KERN_SRC} ${KERN_HDR}
            DEPENDS ${KERN_SRC} ${EMBED_KERNEL_SCRIPT}
            COMMENT "Generate ${KERN_HDR}"
        )

        target_sources(${TARGET_NAME} PRIVATE ${KERN_HDR})
    endif ()
endfunction()

# 首批 kernel: add (S5 起按需扩充; 全部 .cl 源必须在启动期编译, 变体数量收敛)
set(GGML_OCL_KERNELS
    add/add
    set_rows/set_rows
    get_rows/get_rows
    mul_mat/gemv
    mul_mat/gemv_q4_1
    mul_mat/gemm
    mul_mat/gemm_q4_1
    softmax/softmax
    rmsnorm/rmsnorm
    glu/glu
)

foreach (K ${GGML_OCL_KERNELS})
    ggml_ocl_add_kernel(${K})
endforeach()
