find_package(OpenCL REQUIRED)
find_package(Python3 REQUIRED)

set(TARGET_NAME ggml-ocl)

ggml_add_backend_library(${TARGET_NAME}
                         ggml-ocl.cpp
                         ggml-ocl-caps.cpp
                         ggml-ocl-mem.cpp
                         ggml-ocl-kernels.cpp
                         ggml-ocl-exec.cpp
                         ggml-ocl-ops.cpp
                         ops/add.cpp
                         ops/cpy.cpp
                         ops/set_rows.cpp
                         ops/get_rows.cpp
                         ops/mul_mat/mul_mat.cpp
                         ops/mul_mat/gemv_f16_f16.cpp
                         ops/mul_mat/gemv_q4_1_f16.cpp
                         ops/mul_mat/gemm_f16_f16.cpp
                         ops/mul_mat/gemm_q4_1_f16.cpp
                         ops/softmax.cpp
                         ops/rmsnorm.cpp
                         ops/glu.cpp
                         ops/mul.cpp
                         ops/rope.cpp
                         ../../include/ggml-ocl.h)
target_link_libraries(${TARGET_NAME} PRIVATE ${OpenCL_LIBRARIES})
target_include_directories(${TARGET_NAME} PRIVATE ${OpenCL_INCLUDE_DIRS})
target_include_directories(${TARGET_NAME} PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})

option(GGML_OCL_PROFILING "OpenCL profiling (CPU overhead)" OFF)

add_compile_definitions(GGML_OCL_TARGET_VERSION=${GGML_OCL_TARGET_VERSION})

if (GGML_OCL_PROFILING)
    add_compile_definitions(GGML_OCL_PROFILING)
endif ()

# fp16 存储是固定策略: 每个算子只保留 half 版 kernel
set(GGML_OCL_KERNELS
    cpy/cpy
    rope/rope
    add/kernel_add_f16
    mul/mul_f16
    glu/swiglu_f16
    rmsnorm/rmsnorm_f16
    softmax/softmax_f16
    get_rows/get_rows_f16_i32_f16
    set_rows/set_rows_f16_i64_f16
    mul_mat/gemv_q4_1_f16
    mul_mat/gemv_f16_f16
    mul_mat/gemm_q4_1_f16
    mul_mat/gemm_f16_f16
)

# kernel 注册表由上面的列表生成: 新增 kernel 只改这一处
set(EMBED_KERNEL_SCRIPT "${CMAKE_CURRENT_SOURCE_DIR}/kernels/embed_kernel.py")
set(OCL_KERNEL_DIR      "${CMAKE_CURRENT_SOURCE_DIR}/kernels")
set(OCL_KERNELS_INC     "${CMAKE_CURRENT_BINARY_DIR}/autogenerated/ocl_kernels.inc")

set(OCL_KERNEL_SRCS "")
foreach (K ${GGML_OCL_KERNELS})
    list(APPEND OCL_KERNEL_SRCS "${OCL_KERNEL_DIR}/${K}.cl")
endforeach()

file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/autogenerated")
target_include_directories(${TARGET_NAME} PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/autogenerated")

add_custom_command(
    OUTPUT  ${OCL_KERNELS_INC}
    COMMAND ${Python3_EXECUTABLE} ${EMBED_KERNEL_SCRIPT}
            --kernels ${OCL_KERNEL_DIR}
            --out ${OCL_KERNELS_INC}
            ${OCL_KERNEL_SRCS}
    DEPENDS ${OCL_KERNEL_SRCS} ${EMBED_KERNEL_SCRIPT}
    COMMENT "Generate ${OCL_KERNELS_INC}"
    VERBATIM
)

target_sources(${TARGET_NAME} PRIVATE ${OCL_KERNELS_INC})
